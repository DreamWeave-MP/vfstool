Every function and method of `@dream/vfs` carries a Luau signature. `plan.type_definitions()`
renders them, with those of every other extension in the plan, as one `.d.luau` text for an
editor's language server or for Luau's type checker; [Embedding
Luau](@/docs/luau-hosts.md#types-for-editors-and-checks) shows how to get it. vfstool_lib's own
tests check the definitions with Luau's checker and type-check a strict script against them.

This page is what a plan of dream-path's `PathExtension` and `VfsExtension` generates, with the
`lua-write` and `lua-host` features on, copied from the output. It begins:

```text
-- Generated by l3i from the runtime plan; do not edit.
-- Extensions: dream.net, dream.path, dream.vfs
```

`dream.net` is l3i's network bridge, which every plan carries, and `dream.vfs` requires
`dream.path`. Their declarations are left out here: the `dream_net` types come first in the
output, and the `@dream/net` and `@dream/path` modules between the two parts below. [dream-path's
pages](https://dreamweave-mp.github.io/dream_path/docs/luau/) describe its module.

## Userdata, views and streams

Each type is named after its userdata name, with dots as underscores: `dream.vfs.VFS` is
`dream_vfs_VFS`. The comment above each says who owns it and whether the plan gave it a tag.

```lua
    function diffReport(self, sourceA: string, sourceB: string): { sourceA: string, sourceB: string, higherPriority: string, shared: { string }, onlyInA: { string }, onlyInB: { string } }
end

-- dream.vfs.Reader (owned by dream.vfs; tag 5)
-- A sequential reader over a file: a position over the map or bytes its VfsFile keeps.
declare extern type dream_vfs_Reader with
    -- The next length bytes as a new buffer, fewer at the end; advances past them.
    function read(self, length: number): buffer
    -- Copies the next bytes into the buffer at bufferOffset (default 0), at most length (default the space left); returns the count and advances past them.
    function readInto(self, buffer: buffer, bufferOffset: number?, length: number?): number
    -- Moves to position, 0 to size inclusive.
    function seek(self, position: number)
    -- Moves count bytes forward, at most to the end.
    function skip(self, count: number)
    function tell(self): number
    function size(self): number
    -- Releases the backing; every later read on this reader is an error.
    function close(self)
end

-- dream.vfs.Snapshot (owned by dream.vfs; untagged)
-- A directory content-hash baseline for runFinalize and changedFiles.
declare extern type dream_vfs_Snapshot with
    function len(self): number
end

-- dream.vfs.Tree (owned by dream.vfs; untagged)
-- A sorted directory tree: for _, row in tree yields { dir = string, file = VfsFile }; toTable() is the nested { files, subdirs } shape.
declare extern type dream_vfs_Tree with
    -- The nested tree as plain tables, the shape the previous binding returned.
    function toTable(self): { [string]: any }
    function count(self): number
    function __iter(self): (({}, number) -> (number?, { dir: string, file: dream_vfs_VfsFile }), {}, number)
end

-- dream.vfs.VFS (owned by dream.vfs; tag 6)
-- A virtual file system: provider stacks low to high priority and the cached resolved winner per key.
declare extern type dream_vfs_VFS with
    function len(self): number
    function isEmpty(self): boolean
    -- Every resolved key, sorted, as a sequence view; sorted once and shared until the next mutation.
    function keys(self): dream_vfs_Keys
    -- Every resolved entry sorted by key, as a sequence view of { key, file } rows.
    function entries(self): dream_vfs_Entries
    -- The winning file for a path in any spelling, or nil.
    function getFile(self, path: string): dream_vfs_VfsFile?
    function contains(self, path: string): boolean
    -- The files whose key matches a case-insensitive regex, as a tree.
    function findByRegex(self, pattern: string, relative: boolean?): dream_vfs_Tree
    function remaining(self, filterPath: string, replacementsOnly: boolean, allDirs: { string }, relative: boolean?): dream_vfs_Tree
    -- Entries whose key contains the substring (folded like a path), sorted by key.
    function pathsMatching(self, substring: string): dream_vfs_Entries
    -- Entries at or under a directory, matched on component boundaries, sorted by key.
    function pathsWith(self, prefix: string): dream_vfs_Entries
    -- Replaces the whole provider stack for key with one loose winner; returns the previous winner.
    function setWinnerLooseFile(self, key: string, physicalPath: string): dream_vfs_VfsFile?
    function setWinnerFile(self, key: string, file: dream_vfs_VfsFile): dream_vfs_VfsFile?
    -- Inserts every loose file under the directory as a higher-priority provider.
    function pushDirectory(self, path: string)
    -- Inserts a provider at highest priority; false for an unsafe key.
    function pushProvider(self, key: string, provider: dream_vfs_VfsProvider): boolean
    -- Inserts every entry of an archive as the newest, highest-priority source.
    function pushArchive(self, path: string): boolean
    -- Removes only the current winner, revealing the next provider.
    function removeWinner(self, key: string): dream_vfs_VfsProvider?
    -- Removes the key and its whole provider stack.
    function removeResolvedFile(self, key: string): dream_vfs_VfsFile?
    function removeProviderPrefix(self, prefix: string): { { key: string, provider: { source: { path: string, kind: string }, file: dream_vfs_VfsFile } } }
    function removeResolvedPrefix(self, prefix: string): { { key: string, file: dream_vfs_VfsFile } }
    -- Removes every provider of key whose source path equals sourcePath lexically.
    function removeProvider(self, key: string, sourcePath: string): { { source: { path: string, kind: string }, file: dream_vfs_VfsFile } }
    function removeSource(self, sourcePath: string): { { key: string, provider: { source: { path: string, kind: string }, file: dream_vfs_VfsFile } } }
    function removeResolvedMatchingGlob(self, glob: string): { { key: string, file: dream_vfs_VfsFile } }
    -- The sorted directory tree as a stream of { dir, file } rows; tree:toTable() is the nested shape.
    function tree(self, relative: boolean?): dream_vfs_Tree
    function display(self, relative: boolean?): string
    -- Writes every winner into dir; returns how many were written and the ones left out.
    function dumpToDirectory(self, dir: string, useHardlinks: boolean): (number, { { kind: string, key: string, archive: string?, source: string?, error: string? } })
    -- Writes the VFS into dest as one directory; returns the winners left out.
    function collapseInto(self, dest: string, options: { allowCopying: boolean?, extractArchives: boolean?, useSymlinks: boolean? }?): { { kind: string, key: string, archive: string?, source: string?, error: string? } }
    function extractFile(self, vfsPath: string, destDir: string): string?
    function diffDirectory(self, dir: string): { conflicts: { { key: string, incoming: dream_vfs_VfsFile, current: dream_vfs_VfsFile } }, additions: { { key: string, file: dream_vfs_VfsFile } }, unreadable: { { path: string, error: string } } }
    -- Provider report rows for a key, low to high priority, as a sequence view.
    function providerRecordsFor(self, path: string): dream_vfs_ProviderRecords
    -- Why a path resolves to its winner: the winner and the providers it overrides.
    function explain(self, path: string): { key: string, winner: { sourceIndex: number, source: { path: string, kind: string }, key: string, originalPath: string, resolvedPath: string }, overridden: { { sourceIndex: number, source: { path: string, kind: string }, key: string, originalPath: string, resolvedPath: string } } }?
    -- Every provider for a key, low to high priority, as a sequence view of { source, file } rows; nil for an unknown key.
    function providersFor(self, key: string): dream_vfs_Providers?
    function duplicates(self, pattern: string?): { entries: { { key: string, providers: { { sourceIndex: number, source: { path: string, kind: string }, key: string, originalPath: string, resolvedPath: string } }, winnerIndex: number } } }
    function archives(self): { { sourceIndex: number, path: string, entryCount: number, winningEntryCount: number } }
    function archiveEntries(self, archive: string): { { key: string, archivePath: string, originalPath: string, wins: boolean } }
    function filesFromArchive(self, archive: string): dream_vfs_Keys
    function sourceContributions(self): { sources: { { [string]: any } } }
    function materializationPlan(self, dest: string, options: { allowCopying: boolean?, extractArchives: boolean?, useSymlinks: boolean? }?): { actions: { { [string]: string? } }, issues: { { [string]: string? } } }
    function layerIndex(self): dream_vfs_LayerIndex
    function serializeTree(self, relative: boolean?, format: string): string
    -- The directory writes go under, as the canonical path, or nil when this VFS refuses writes.
    function writeRoot(self): string?
    -- Makes dir the write root, created if absent and kept as its canonical path. Needs the filesystem.write capability.
    function setWriteRoot(self, dir: string)
    -- Writes data to <writeRoot>/<key>, creating directories, then makes that file the key's winner and returns it. offset writes in place without truncating; append adds at the end; create = false refuses a file that does not exist.
    function writeFile(self, key: string, data: buffer | string, options: { offset: number?, append: boolean?, create: boolean? }?): dream_vfs_VfsFile
    -- A writer over <writeRoot>/<key>, truncated unless append or truncate = false; close makes the file the key's winner.
    function openWrite(self, key: string, options: { append: boolean?, truncate: boolean? }?): dream_vfs_Writer
    -- Creates the directory <writeRoot>/<key> and its parents.
    function mkdir(self, key: string)
    -- Deletes <writeRoot>/<key>, a file or a whole directory, and drops the winners it held; a key with nothing under the write root is an error.
    function remove(self, key: string)
    -- Moves <writeRoot>/<from> to <writeRoot>/<to>, a file or a directory, and moves the winners with it.
    function rename(self, from: string, to: string)
end

-- dream.vfs.VfsFile (owned by dream.vfs; tag 7)
-- A loose file on disk or an entry inside an archive.
declare extern type dream_vfs_VfsFile with
    function isLoose(self): boolean
    function isArchive(self): boolean
    -- The host path of a loose file, or the entry name inside its archive, byte for byte.
    function path(self): string
    function fileName(self): string?
    function fileStem(self): string?
    function parentArchivePath(self): string?
    function parentArchiveName(self): string?
    -- The size in bytes: a loose file's metadata, or an archive entry's uncompressed size, from the archive's index when it has one.
    function size(self): number
    -- The whole file as a string; readInto is the copy-free form.
    function readAll(self): string
    -- The whole file as a new buffer, one copy from the file's map or bytes.
    function readAllBuffer(self): buffer
    -- Reads the file into the buffer from offset (default 0), at most the space left; returns the bytes written.
    function readInto(self, buffer: buffer, offset: number?): number
    -- Copies length bytes (default the space left) from fileOffset into the buffer at bufferOffset (default 0); returns the count, fewer only at the end of the file.
    function readAt(self, buffer: buffer, fileOffset: number, length: number?, bufferOffset: number?): number
    -- A new buffer of length bytes from fileOffset, shorter only at the end of the file.
    function readRange(self, fileOffset: number, length: number): buffer
    -- A sequential reader over the file's map or bytes.
    function open(self): dream_vfs_Reader
end

-- dream.vfs.VfsLock (owned by dream.vfs; untagged)
-- A deterministic lock manifest of the current winners.
declare extern type dream_vfs_VfsLock with
    function schemaVersion(self): number
    function entries(self): { { key: string, winnerSource: string, winnerKind: string, winnerHashBlake3: string?, winnerSize: integer?, providerCount: number } }
    function toTable(self): { schemaVersion: number, entries: { { key: string, winnerSource: string, winnerKind: string, winnerHashBlake3: string?, winnerSize: integer?, providerCount: number } } }
end

-- dream.vfs.VfsProvider (owned by dream.vfs; untagged)
-- One provider for a key: its source and backing file.
declare extern type dream_vfs_VfsProvider with
    function source(self): { path: string, kind: string }
    function file(self): dream_vfs_VfsFile
end

-- dream.vfs.Writer (owned by dream.vfs; untagged)
-- A buffered writer over a file; close registers a VFS writer's file as its key's winner.
declare extern type dream_vfs_Writer with
    -- Writes data (a buffer or a string), or its slice from offset of length bytes, at the position and advances past it; returns the count.
    function write(self, data: buffer | string, offset: number?, length: number?): number
    -- Writes data at position without moving; an error on an append writer.
    function writeAt(self, position: number, data: buffer | string, offset: number?, length: number?): number
    -- Moves to position; past the end, the next write extends the file.
    function seek(self, position: number)
    function tell(self): number
    function flush(self)
    -- Cuts or extends the file to length bytes; a position past it moves to it.
    function truncate(self, length: number)
    -- Flushes and closes. A writer from vfs:openWrite then makes its file the key's winner and returns it; closing again does nothing.
    function close(self): dream_vfs_VfsFile?
end
```

## The module

The module's own type. Report tables are spelled out inline. The last line appears only when the
host exposes the [`vfstool` global](@/docs/luau-hosts.md#the-vfstool-global).

```lua
-- module @dream/vfs (provided by dream.vfs)
-- vfstool: OpenMW-style virtual file systems, provider analysis, and run workflows.
export type Module__dream_vfs = {
    -- Backslashes to slashes and ASCII lowercase, for textual host path comparisons.
    normalizeHostPath: (path: string) -> string,
    -- The same as normalizeHostPath; Luau strings are immutable, so the name is kept for compatibility only.
    normalizeHostPathInPlace: (path: string) -> string,
    pathGlobMatches: (glob: string, path: string) -> boolean,
    sourceGlobMatches: (glob: string, sourcePath: string) -> boolean,
    analyzePair: (path: string, leftBytes: string, rightBytes: string) -> { assetClass: string, delta: { kind: string, changeSummary: { string }? } },
    -- Dumps the VFS into mergedDir and snapshots it. Hardlinks let child tools edit the originals in place.
    runSetup: (vfs: dream_vfs_VFS, mergedDir: string, useHardlinks: boolean) -> (number, dream_vfs_Snapshot),
    runSetupTracked: (vfs: dream_vfs_VFS, mergedDir: string, useHardlinks: boolean) -> (number, dream_vfs_MetadataSnapshot),
    runFinalize: (mergedDir: string, outputDir: string, snapshot: dream_vfs_Snapshot) -> { { relativePath: string, destinationPath: string } },
    runFinalizeTracked: (mergedDir: string, outputDir: string, snapshot: dream_vfs_MetadataSnapshot) -> { { relativePath: string, destinationPath: string } },
    snapshotDirectory: (dir: string) -> dream_vfs_Snapshot,
    snapshotDirectoryMetadata: (dir: string) -> dream_vfs_MetadataSnapshot,
    changedFiles: (dir: string, snapshot: dream_vfs_Snapshot) -> { string },
    changedFilesMetadata: (dir: string, snapshot: dream_vfs_MetadataSnapshot) -> { string },
    -- Encodes a value as json, yaml, or toml; tables, sequence views, strings, numbers, booleans.
    serialize: (value: any, format: string) -> string,
    -- VFS constructors.
    VFS: { new: () -> dream_vfs_VFS, fromDirectories: (dirs: { string }, options: { archives: { string }?, writeRoot: string? }?) -> dream_vfs_VFS, fromDirectoriesWithConflictIndex: (dirs: { string }, options: { archives: { string }?, writeRoot: string? }?) -> (dream_vfs_VFS, dream_vfs_ConflictIndex), fromDirectoriesWithLayerIndex: (dirs: { string }, options: { archives: { string }?, writeRoot: string? }?) -> (dream_vfs_VFS, dream_vfs_LayerIndex) },
    VfsFile: { from: (path: string) -> dream_vfs_VfsFile },
    VfsProvider: { new: (source: { path: string, kind: string }, file: dream_vfs_VfsFile) -> dream_vfs_VfsProvider },
    LayerIndex: { fromFileLists: (sources: { { source: { path: string, kind: string }, files: { string } } }) -> dream_vfs_LayerIndex },
    ConflictIndex: { fromDirectories: (dirs: { string }) -> dream_vfs_ConflictIndex, fromFileLists: (sources: { { source: string, files: { string } } }) -> dream_vfs_ConflictIndex, fromLayerIndex: (layer: dream_vfs_LayerIndex) -> dream_vfs_ConflictIndex },
    -- Files and directories on host paths, with no VFS in between (the lua-host feature); every function needs the filesystem.host capability.
    host: { readFile: (path: string) -> buffer, readFileString: (path: string) -> string, readAt: (path: string, offset: number, length: number) -> buffer, writeFile: (path: string, data: buffer | string, options: { append: boolean? }?) -> number, open: (path: string) -> dream_vfs_Reader, openWrite: (path: string, options: { append: boolean?, truncate: boolean? }?) -> dream_vfs_Writer, stat: (path: string) -> { size: number, isFile: boolean, isDir: boolean, modified: number?, readonly: boolean }?, exists: (path: string) -> boolean, list: (path: string, options: { recursive: boolean? }?) -> dream_vfs_HostEntries, mkdir: (path: string, options: { recursive: boolean? }?) -> (), remove: (path: string, options: { recursive: boolean? }?) -> (), rename: (from: string, to: string) -> (), copy: (from: string, to: string) -> number, canonicalize: (path: string) -> string },
}
declare vfstool: Module__dream_vfs
```
